// ЗАДАНИЕ №1
// Интерфейс
interface IStack<T> {
  items: T[];
  push(item: T): void;
  pop(): T | undefined;
  isEmpty(): boolean;
  size(): number;
}

// Класс
class Stack<T> implements IStack<T> {
  items: T[];
  constructor() {
    this.items = [];
  }

  // push - добавляет элемент в стек
  push(item: T): void {
    this.items.push(item);
  }

  // pop - удаляет и возращает последний добавленный элемент стека. Если стек пустой, возвращает undefined
  pop(): T | undefined {
    return this.items.pop();
  }

  // isEmpty - возвращает true, если стек пустой, и false в противном случае
  isEmpty(): boolean {
    return this.items.length === 0;
  }

  // size - возвращает количество элементов в стеке
  size(): number {
    return this.items.length;
  }
}

// Создание стэков (массивов или башен) с типами данных number

// Рекурсивная функция
// Параметры: (n, stack1, stack2, stack3)
// n - количество вызовов функции
// stack1 - основной стэк (массив или башня)
// stack2 - вспомогательный стэк (массив или башня)
// stack3 - целевой стэк (массив или башня)
function movingStacks(n: number, source: IStack<number | undefined>, spare: IStack<number | undefined>, target: IStack<number | undefined>): void {
  if (n === 1) {
    target.push(source.pop());
    console.log(stack1.items, stack2.items, stack3.items);
    return;
  }
  console.log(stack1.items, stack2.items, stack3.items);

  movingStacks(n - 1, source, target, spare);

  target.push(source.pop());

  movingStacks(n - 1, spare, source, target);
}

// При каждом вызове рекурсивной функции самой себя процесс погружается глубже на один уровень до тех пор, пока n не будет равняться 0

// TASK 1
type Dinamic1 = Record<string, string>;
type Dinamic2 = { [key: string]: number; };
type Tuple = [string, number, string];

interface Car {
  model: string;
  price: number;
  dynamic_1: Dinamic1;
  dynamic_2: Dinamic2;
  tuple: Tuple;
}

class RandomCar implements Car {
  model: string;
  price: number;
  dynamic_1: Dinamic1;
  dynamic_2: Dinamic2;
  tuple: Tuple;

  constructor(model: string, price: number, dynamic_1: Dinamic1, dynamic_2: Dinamic2, tuple: Tuple) {
    this.model = model;
    this.price = price;
    this.dynamic_1 = dynamic_1;
    this.dynamic_2 = dynamic_2;
    this.tuple = tuple;
  }
}
